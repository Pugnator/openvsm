find_package(Doxygen REQUIRED)

set(DOXYGEN_PROJECT_NAME "OpenVSM")
set(DOXYGEN_PROJECT_BRIEF "Lua-based model creation for Proteus VSM")
set(DOXYGEN_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "${CMAKE_CURRENT_SOURCE_DIR}/DEVELOPER-GUIDE.md")
set(DOXYGEN_EXTRACT_ALL YES)
set(DOXYGEN_EXTRACT_PRIVATE YES)
set(DOXYGEN_RECURSIVE YES)
set(DOXYGEN_SOURCE_BROWSER YES)
set(DOXYGEN_INLINE_SOURCES YES)
set(DOXYGEN_GENERATE_TREEVIEW YES)
set(DOXYGEN_WARN_IF_UNDOCUMENTED YES)
set(DOXYGEN_WARN_IF_DOC_ERROR YES)
set(DOXYGEN_QUIET YES)

doxygen_add_docs(
  openvsm-docs
  "${CMAKE_SOURCE_DIR}/README.md"
  "${CMAKE_CURRENT_SOURCE_DIR}/DEVELOPER-GUIDE.md"
  "${CMAKE_CURRENT_SOURCE_DIR}/CODE-STYLE.md"
  "${CMAKE_SOURCE_DIR}/model/cpp"
  COMMENT "Building the OpenVSM developer reference"
)
